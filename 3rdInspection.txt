========== DATASET PREPARATION SCRIPT ==========
"""
build_dataset.py
----------------
Builds a machine learning training dataset from:
1. mwccc_files.csv (682 real Malawian court cases)
2. Available MWCC XML transcripts
3. Statutory judicial urgency rules (Ground Truth Proxy)
"""

import os
import re
import xml.etree.ElementTree as ET
import pandas as pd

# Paths
BASE_DIR = os.getcwd()
CSV_PATH = os.path.join(BASE_DIR, "Datasets", "mwccc_files.csv")
XML_DIR = os.path.join(BASE_DIR, "Datasets", "files-archive (2)", "MWCC Corpus EXTRACTS-20210902", "MWCC Corpus EXTRACTS")
OUTPUT_PATH = os.path.join(BASE_DIR, "Datasets", "dataset_for_modeling.csv")


def extract_xml_text(file_id, year):
    """If an XML file exists for this case, extract its raw text."""
    xml_path = os.path.join(XML_DIR, str(year), f"{file_id}.xml")
    if not os.path.exists(xml_path):
        return None
    try:
        tree = ET.parse(xml_path)
        root = tree.getroot()
        paragraphs = []
        for p in root.iter():
            tag = p.tag.split("}")[-1] if "}" in p.tag else p.tag
            if tag == "p" and p.text:
                paragraphs.append(p.text.strip())
        return " ".join(paragraphs) if paragraphs else None
    except Exception:
        return None


def parse_raw_title(raw_title, file_id):
    """
    Cleans raw file title into structured metadata:
    - Parties (e.g. 'Lawrence Chibwana Vs The State')
    - Case Type (e.g. 'Criminal Appeal')
    - Case Number (e.g. 'Criminal Appeal No 42 of 2010')
    - Year
    - Legal tags (e.g. 'Defilement', 'Cashgate', 'Murder')
    """
    clean = raw_title.replace("_", " ")

    # 1. Extract Year from file_id (e.g., judg_2010_1 -> 2010)
    year_match = re.search(r"judg_(\d{4})", file_id)
    year = int(year_match.group(1)) if year_match else 2015

    # 2. Extract Case Number
    num_match = re.search(
        r"(?:Case|Cause|Appeal|App)\s*(?:No\.?|Number)?\s*(\d+\s*(?:of|/)\s*\d{4})",
        clean,
        re.IGNORECASE,
    )
    case_number = num_match.group(0).strip() if num_match else f"Case {file_id}"

    # 3. Detect Case Type
    lower = clean.lower()
    if "bail" in lower:
        case_type = "Bail Application"
    elif "murder" in lower or "homicide" in lower:
        case_type = "Homicide / Murder"
    elif "appeal" in lower:
        case_type = "Criminal Appeal"
    elif "confirmation" in lower:
        case_type = "Confirmation Case"
    elif "review" in lower:
        case_type = "Criminal Review"
    elif "miscellaneous" in lower or "misc" in lower:
        case_type = "Miscellaneous Criminal"
    else:
        case_type = "Criminal Case"

    # 4. Extract Parties (clean up before the double-underscore)
    parties = clean.split("  ")[0]
    if "No" in parties:
        parties = re.split(r"(?:Criminal|Confirmation|Bail|Misc)", parties)[0]
    parties = parties.strip()
    if len(parties) < 3:
        parties = clean[:60]

    return {
        "case_name": parties,
        "case_type": case_type,
        "year": year,
        "case_number": case_number,
        "clean_title": clean,
    }


def generate_synthesized_text(meta, full_text):
    """
    If we have the actual XML transcript, we use it!
    Otherwise, we reconstruct a legal filing summary based on the statutory
    particulars extracted from the title.
    """
    if full_text and len(full_text.strip()) > 50:
        return full_text

    # Reconstruct legal filing narrative
    clean = meta["clean_title"]
    c_type = meta["case_type"]
    year = meta["year"]
    parties = meta["case_name"]

    narrative = [
        f"In the High Court of Malawi, Principal Registry.",
        f"Matter: {parties}.",
        f"Filing Category: {c_type}, Registered in {year}.",
    ]

    lower = clean.lower()
    if "bail" in lower:
        narrative.append(
            "Application for admission to bail pending trial. The applicant remains in custody "
            "on remand at prison. Submissions made regarding deprivation of personal liberty, "
            "constitutional rights under section 42, and absence of flight risk."
        )
    elif "murder" in lower or "homicide" in lower:
        narrative.append(
            "Indictment for murder contrary to section 209 of the Penal Code. Capital offense proceedings. "
            "Accused remanded in custody awaiting trial. State witnesses summoned."
        )
    elif "defilement" in lower or "minor" in lower or "child" in lower:
        narrative.append(
            "Offense involving vulnerable victim, defilement of girl under 16 years. Urgent child protection "
            "and statutory custody priority."
        )
    elif "cashgate" in lower or "theft" in lower or "fraud" in lower:
        narrative.append(
            "High-impact public financial crime and embezzlement of government funds. Significant public interest "
            "and economic governance implications."
        )
    elif "appeal" in lower:
        narrative.append(
            "Appeal against conviction and custodial sentence delivered by the subordinate magistrate court. "
            "Appellant has served substantial portion of prison term awaiting determination of appeal."
        )
    elif "confirmation" in lower:
        narrative.append(
            "Sentencing confirmation referred from the resident magistrate court for High Court review "
            "pursuant to the Criminal Procedure and Evidence Code."
        )
    else:
        narrative.append(
            "General criminal cause scheduled for directions, plea taking, and hearing before the High Court."
        )

    return " ".join(narrative)


def assign_priority_label(case_type, year, text):
    """
    Supervised Learning Ground Truth Logic:
    Rules derived from Malawi Judiciary Criminal Practice Directions:
    1. HIGH: Liberty/custody issues (Bail), Capital offenses (Murder), vulnerable parties (defilement/minors),
             or extreme delay (older than 10 years).
    2. MEDIUM: Substantive appeals, sentencing confirmations, significant fraud.
    3. LOW: Routine reviews, miscellaneous applications.
    """
    lower = text.lower()
    age = 2026 - year

    # High Priority triggers
    is_bail = case_type == "Bail Application" or "bail" in lower or "custody" in lower
    is_capital = case_type == "Homicide / Murder" or "murder" in lower or "homicide" in lower
    is_vulnerable = "defilement" in lower or "minor" in lower or "child" in lower
    is_chronic_backlog = age >= 12  # Unresolved for > 12 years

    if is_bail or is_capital or is_vulnerable or is_chronic_backlog:
        return "High"

    # Medium Priority triggers
    is_appeal = case_type in ["Criminal Appeal", "Confirmation Case"]
    is_moderate_backlog = age >= 8

    if is_appeal or is_moderate_backlog or "cashgate" in lower:
        return "Medium"

    # Low Priority
    return "Low"


def main():
    print("=" * 60)
    print("BUILDING DATASET FOR MACHINE LEARNING")
    print("=" * 60)

    # 1. Read the 682 indexed cases
    df_raw = pd.read_csv(CSV_PATH, header=None, names=["raw_title", "file_id"])
    print(f"Loaded {len(df_raw)} cases from mwccc_files.csv")

    records = []
    xml_matches = 0

    for _, row in df_raw.iterrows():
        raw_title = str(row["raw_title"])
        file_id = str(row["file_id"]).strip()

        meta = parse_raw_title(raw_title, file_id)

        # Check if full text XML is available
        full_text = extract_xml_text(file_id, meta["year"])
        if full_text:
            xml_matches += 1

        # Synthesize/augment text
        text_content = generate_synthesized_text(meta, full_text)

        # Compute ground truth priority label
        priority = assign_priority_label(meta["case_type"], meta["year"], text_content)

        records.append(
            {
                "case_id": file_id,
                "case_name": meta["case_name"],
                "case_type": meta["case_type"],
                "year": meta["year"],
                "court_level": "High Court of Malawi",
                "case_number": meta["case_number"],
                "text_content": text_content,
                "priority_tier": priority,
            }
        )

    df_out = pd.DataFrame(records)
    df_out.to_csv(OUTPUT_PATH, index=False)

    print(f"\nSuccessfully generated: {OUTPUT_PATH}")
    print(f"Total cases processed: {len(df_out)}")
    print(f"Matched full XML transcripts: {xml_matches}")

    print("\n--- PRIORITY TIER DISTRIBUTION (Target Label y) ---")
    print(df_out["priority_tier"].value_counts())
    print("\nPercentages:")
    print(df_out["priority_tier"].value_counts(normalize=True).round(3) * 100)

    print("\n--- CASE TYPE BREAKDOWN ---")
    print(df_out["case_type"].value_counts())
    print("=" * 60)


if __name__ == "__main__":
    main()

========== MODULE DESCRIPTION ==========
JUDICIAL CASE PRIORITIZATION SYSTEM
MODULE DESCRIPTION

Purpose
-------
This document describes the Flask reference application in the jcp/ directory.
It is based on the source code currently in this repository. The system is an
administrative scheduling-support prototype: its priority score is not a
finding about guilt, merits, or the correct judicial outcome. A human court
official remains responsible for case-management decisions.


1. Application startup and configuration — jcp/app.py
------------------------------------------------------
The Flask application is the web entry point. Run it from the jcp/ directory
with:

    python app.py

At startup, app.py configures Flask, Flask-Login, CSRF protection, SQLAlchemy,
the upload-size limit, and proxy/cookie behavior. It creates database tables
and seeds demo users, judges, and cases if the database has no users.

The database URL can be set with DATABASE_URL; otherwise the application uses
SQLite (normally jcp/instance/jcps.db). SECRET_KEY, PORT, FLASK_DEBUG,
IDLE_LOCK_SECONDS, MODEL_PATH, and ASSET_V also affect runtime configuration.
Set a unique secret and disable debug mode in any non-local deployment.

Authentication and access control:
- Flask-Login protects the docket and case-management pages.
- The application defines clerk, registrar, and admin roles.
- The manager_required decorator restricts case overrides and bulk status
  changes to users whose role is registrar or admin.
- Registration currently assigns the registrar role in app.py. Review this
  policy before exposing registration in a live deployment.
- An inactivity lock can return a locked page or HTTP 423 for fetch requests.
- CSRF protection applies to form submissions.

Main routes and responsibilities:
- / and /dashboard: searchable, filterable, sortable, grouped, paginated docket.
- /case/<id>: case detail.
- /case/<id>/update: update workflow fields such as status, hearing date, and
  judge assignment.
- /case/<id>/override: record a human priority override with a reason.
- /new: create and score a single case.
- /upload: import cases from a UTF-8 CSV and score each accepted row.
- /template.csv: download a CSV header and example row.
- /cases/export: export selected cases, or all cases when none are selected.
- /cases/bulk-status: apply a workflow status to selected cases.
- /model: show model information and aggregate scoring/override statistics.
- /audit: view the audit trail.
- /api/notifications: return recent case-related audit events as JSON.
- /login, /register, /logout, /lock, /unlock: authentication and session-lock
  flows.

CSV import behavior:
- The importer requires the case_id and case_name headers. The other template
  columns are case_type, year, court_level, case_number, and text_content.
- Each row must have a non-empty case_id and case_name. Existing case IDs are
  counted as skipped. Accepted rows are scored and saved.
- A file that omits either required header is rejected before row processing.
- Year values that are not numeric are treated as missing by the current
  importer. The importer does not accept a priority, status, hearing date, or
  judge from the CSV; the model assigns the initial tier and the workflow
  status uses its default.
- Import reports distinguish accepted rows, duplicate IDs, and row errors.


2. Data model and persistence — jcp/models.py
---------------------------------------------
SQLAlchemy models define the persistent entities:

- User: username, email, password hash, role, and creation time. Passwords are
  hashed using Werkzeug. The can_manage property identifies registrar/admin
  users.
- Judge: judge name, division, and related cases.
- Case: unique case ID, case caption, type, filing/judgment year, court level,
  case number, and source text. It also stores model tier, confidence,
  explanation, explanation contributions, and model version.
- Case human-decision fields: override tier, reason, user, and timestamp.
- Case workflow fields: status, hearing date, and assigned judge.
- Audit: timestamped user actions such as login, import, create, export,
  assignment, status change, and priority override.

Case helper properties expose the effective tier (human override first),
whether a low-confidence result needs review, a queue group key, and an
age label. The review threshold is 0.55 confidence. The application
initializes database tables and seed data from app.py.


3. Prediction and explanations — jcp/ml.py
-----------------------------------------
The prediction module accepts case metadata and text and returns a priority
tier, confidence, class probabilities, explanation contributions, a narrative,
and model/version metadata.

When MODEL_PATH points to a compatible scikit-learn pipeline, the module loads
the pickle and calls predict_proba. It can use SHAP TreeExplainer when SHAP is
available; otherwise it approximates contributions using feature importance
and feature activation. The expected model pipeline ends in a tree estimator.

If no model file is available, a deterministic keyword-based heuristic keeps
the prototype usable. It looks for terms related to custody/remand, bail,
serious offences, appeals, constitutional questions, vulnerable parties, and
procedural or routine matters, and also considers case age. This fallback is
not a trained or validated judicial model. Its outputs are only prototype
administrative suggestions and require human review. Confidence below 0.55 is
marked for review by the case model.

The displayed explanation is generated from the strongest contributions. Model
information returned by info() supports the model-card page.


4. HTML presentation — jcp/templates/
-------------------------------------
Jinja templates render the pages and share a common shell:

- base.html: page layout, navigation, command areas, and shared UI elements.
- dashboard.html: docket queue and its filters/actions.
- case_detail.html: selected case information and management controls.
- new_case.html and upload.html: single-case entry and CSV batch import.
- login.html, register.html, lock.html, and _lock.html: authentication and
  locked-session UI.
- model_card.html: model mode, statistics, and limitations.
- audit.html: audit event list.
- error.html: access-denied and not-found messages.
- _macros.html and the icon sprite in the templates directory: reusable
  template fragments and icons.

Template visibility is not the security boundary: route-level authentication
and authorization in app.py enforce access.


5. Browser behavior and styling — jcp/static/
---------------------------------------------
- static/js/app.js implements client-side shell behavior, theme and navigation
  preferences, flyouts, notifications/toasts, and other UI interactions.
- static/css/custom.css defines the visual system and responsive styling.
- static/fonts/ and static/assets/ (where present) contain local presentation
  assets such as fonts and icons.

The upload page includes client-side file selection checks, but the server-side
checks in app.py remain authoritative.


6. Dependencies — jcp/requirements.txt
--------------------------------------
The application depends on Flask, Flask-Login, Flask-SQLAlchemy, Flask-WTF,
scikit-learn, and NumPy. SHAP is optional and enables TreeExplainer-based
explanations when compatible with the loaded model.


7. Related dataset preparation scripts
--------------------------------------
The repository-root build_dataset.py and inspect_data.py are separate from the
running Flask application. They prepare or inspect modeling data using the
MWCC corpus files in Datasets/. The jcp CSV importer does not run those scripts:
it takes the case fields in an uploaded CSV, scores them with ml.py, and stores
the resulting case records.

The accompanying MWCC import example uses real case captions and corpus file
IDs from Datasets/mwccc_files.csv. Other fields are limited to metadata that
can be derived from those captions/IDs or explicitly marked test-fixture text.
It does not assert case facts, current status, priority, or outcome. Importing
any row into the application creates a new docket record with the application's
default workflow status; therefore these rows are for testing and must not be
treated as a verified live backlog.
